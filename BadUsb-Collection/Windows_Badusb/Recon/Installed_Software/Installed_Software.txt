REM Author: UNC0V3R3D
REM Description: Lists all installed software from the registry — both 32bit and 64bit.
REM Version: 1.0
REM Category: Recon
REM Target: Windows 10/11
DELAY 1000
GUI r
DELAY 700
STRING powershell -windowstyle hidden -Command "$paths=@('HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*','HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*','HKCU:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*');Get-ItemProperty $paths -ErrorAction SilentlyContinue | Where-Object {$_.DisplayName} | Select DisplayName,DisplayVersion,Publisher,InstallDate | Sort DisplayName | Format-Table -AutoSize | Out-File $env:TEMP\installed_sw.txt -Force;Start-Process notepad $env:TEMP\installed_sw.txt"
ENTER
DELAY 4000
REM End of payload
