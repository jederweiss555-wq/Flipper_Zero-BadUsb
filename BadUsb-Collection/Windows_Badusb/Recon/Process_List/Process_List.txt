DELAY 1000
GUI r
DELAY 700
STRING powershell -windowstyle hidden -Command "Get-Process | Select-Object Name,Id,CPU,WorkingSet,Path | Sort-Object Name | Format-Table -AutoSize | Out-File $env:TEMP\processes.txt -Force; Start-Process notepad $env:TEMP\processes.txt"
ENTER
DELAY 3000
REM End of payload
